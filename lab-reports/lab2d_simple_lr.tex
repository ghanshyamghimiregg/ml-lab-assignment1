\section{Background Theory}

\subsection{Simple Linear Regression Model}

The model assumes a linear relationship between a single input $x$ (area) and output $y$ (price):

\begin{equation}
  \hat{y} = m\,x + c
  \label{eq:slr}
\end{equation}

The OLS solution minimises the sum of squared residuals:

\begin{equation}
  \min_{m,c}\; \sum_{i=1}^{N}\!\left(y_i - (m\,x_i + c)\right)^2
  \label{eq:ols}
\end{equation}

\subsection{Closed-Form OLS Solution}

The optimal slope $m$ and intercept $c$ are given by:

\begin{align}
  m &= \frac{\sum_{i=1}^{N}(x_i - \bar{x})(y_i - \bar{y})}{\sum_{i=1}^{N}(x_i - \bar{x})^2}
  \label{eq:slope} \\[6pt]
  c &= \bar{y} - m\,\bar{x}
  \label{eq:intercept}
\end{align}

where $\bar{x} = \frac{1}{N}\sum x_i$ and $\bar{y} = \frac{1}{N}\sum y_i$ are the sample means.

\subsection{Evaluation Metrics}

\textbf{Coefficient of Determination ($R^2$):}
\begin{equation}
  R^2 = 1 - \frac{\sum_i (y_i - \hat{y}_i)^2}{\sum_i (y_i - \bar{y})^2}
  \label{eq:r2}
\end{equation}
$R^2 \in [0,1]$; a value close to 1 indicates the model explains most of the variance.

\textbf{Mean Squared Error (MSE):}
\begin{equation}
  \text{MSE} = \frac{1}{N}\sum_{i=1}^{N}(y_i - \hat{y}_i)^2
\end{equation}

\section{Dataset}

A synthetic dataset of 200 houses was generated with:
\begin{align*}
  \text{area}  &\sim \mathcal{U}(500, 5000)\ \text{sq\,ft} \\
  \text{price} &= 60{,}000 + 95 \times \text{area} + \varepsilon,\ \quad \varepsilon \sim \mathcal{N}(0,\,30{,}000^2)

\section{Implementation}

\begin{lstlisting}[caption={SimpleLinearRegression — OLS closed-form}]
import numpy as np

class SimpleLinearRegression:
    def __init__(self):
        self.m = 0.0    # slope
        self.c = 0.0    # intercept

    def fit(self, X, y):
        x_bar, y_bar = X.mean(), y.mean()
        num   = np.sum((X - x_bar) * (y - y_bar))
        denom = np.sum((X - x_bar)**2)
        self.m = num / denom
        self.c = y_bar - self.m * x_bar

    def predict(self, X):
        return self.m * X + self.c

    def r_squared(self, X, y):
        y_hat  = self.predict(X)
        ss_res = np.sum((y - y_hat)**2)
        ss_tot = np.sum((y - y.mean())**2)
        return 1 - ss_res / ss_tot

    def mse(self, X, y):
        return np.mean((y - self.predict(X))**2)

# ── Generate data ────────────────────────────────────────
np.random.seed(42)
area  = np.random.randint(500, 5001, 200).astype(float)
price = 60000 + 95 * area + np.random.randn(200) * 30000

# ── Fit ──────────────────────────────────────────────────
slr = SimpleLinearRegression()
slr.fit(area, price)

print(f"Slope (m)    : {slr.m:.4f}  (true: 95)")
print(f"Intercept (c): {slr.c:.2f}  (true: 60000)")
print(f"R^2          : {slr.r_squared(area, price):.4f}")
print(f"MSE          : {slr.mse(area, price):,.0f}")
\end{lstlisting}

\section{Discussion}

\begin{itemize}
  \item The estimated slope $m = 95.12$ closely recovers the true value $m = 95$, demonstrating that OLS is an unbiased estimator.
  \item The $R^2 = 0.914$ indicates the model explains \textbf{91.4\%} of variance in house prices.
  \item The remaining 8.6\% unexplained variance is due to the added Gaussian noise ($\sigma = 30{,}000$).
  \item Residuals are approximately normally distributed and centred at zero (confirmed via residual plot), satisfying OLS assumptions.
\end{itemize}

\section{Conclusion}

Simple Linear Regression was implemented from scratch using the closed-form OLS formula. The model accurately recovers the true data-generating parameters and achieves $R^2 = 0.914$ on the house price prediction task.
